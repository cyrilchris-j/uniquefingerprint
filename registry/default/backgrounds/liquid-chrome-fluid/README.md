# Liquid Chrome Fluid

Interactive viscous metallic fluid canvas reacting to mouse and touch turbulence.

## Installation

```bash
pnpm dlx uniquefingerprint add liquid-chrome-fluid
```

## Usage

```tsx
import { LiquidChromeFluid } from "@/components/liquid-chrome-fluid";

export default function Page() {
  return (
    <LiquidChromeFluid className="min-h-screen flex items-center justify-center">
      <h1 className="text-4xl font-bold text-white relative z-10">
        Future Interface
      </h1>
    </LiquidChromeFluid>
  );
}
```
